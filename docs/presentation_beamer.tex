\section{Architecture HPC}

\begin{frame}{Le « Mur de la Mémoire » et la Réponse de BLAS}
    \begin{columns}[c]
        \begin{column}{0.54\textwidth}
            \begin{messagecle}[\textbf{Le Défi : Le Mur de la Mémoire (\textit{Memory Wall})}]
                \small
                Les processeurs calculent beaucoup plus vite que la RAM ne peut fournir les données. Si le processeur attend la mémoire, les performances s'effondrent.
            \end{messagecle}
            \vspace{0.1cm}
            \begin{alertblock}{La Clé : L'Intensité Arithmétique}
                \small
                $$q = \frac{\text{Nombre d'opérations (FLOPs)}}{\text{Nombre d'octets transférés}}$$
                \begin{itemize}
                    \item \textbf{BLAS 1 \& 2 :} $q = \mathcal{O}(1)$ $\implies$ \textcolor{crimsonred}{\textbf{Bloqué par le bus mémoire}}.
                    \item \textbf{BLAS 3 (\texttt{DGEMM}) :} $q = \mathcal{O}(n)$ $\implies$ \textcolor{forestgreen}{\textbf{Pleine puissance processeur}} !
                \end{itemize}
            \end{alertblock}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{table}
                \centering
                \small
                \begin{tabular}{l c c}
                    \toprule
                    \textbf{Niveau BLAS} & \textbf{Calcul} & \textbf{Routine Clé} \\
                    \midrule
                    \textbf{Niveau 1} & $\mathcal{O}(n)$ & \texttt{DAXPY} \\
                    \textbf{Niveau 2} & $\mathcal{O}(n^2)$ & \texttt{DGEMV} \\
                    \textbf{Niveau 3} & $\mathbf{\mathcal{O}(n^3)}$ & \textbf{\texttt{DGEMM}} \\
                    \bottomrule
                \end{tabular}
            \end{table}

            \vspace{0.1cm}
            \begin{succesbox}[\textbf{Le Secret : Cache Blocking}]
                \scriptsize
                En découpant la matrice en sous-blocs $B \times B$ calibrés pour le cache L1 ($32\text{ Ko}$), chaque coefficient est réutilisé $B$ fois sur place !
            \end{succesbox}
        \end{column}
    \end{columns}
\end{frame}

\section{Modèle Intra-Hôte}

\begin{frame}{Le Modèle Intra-Hôte de Nowak-May-Perelson}
    \begin{columns}[c]
        \begin{column}{0.54\textwidth}
            \begin{messagecle}[\textbf{Les 3 Compartiments Biologiques}]
                \small
                Soit $\eta \in [0, 1]$ l'efficacité du traitement antirétroviral :
                \begin{equation*}
                \left\{
                \begin{aligned}
                    \frac{\mathrm{d}T}{\mathrm{d}t} &= s - dT - (1 - \eta)\beta T V \\[0.06cm]
                    \frac{\mathrm{d}I}{\mathrm{d}t} &= (1 - \eta)\beta T V - \delta I \\[0.06cm]
                    \frac{\mathrm{d}V}{\mathrm{d}t} &= p I - c V
                \end{aligned}
                \right.
                \end{equation*}
            \end{messagecle}
            \vspace{0.08cm}
            \begin{itemize}\footnotesize
                \item $T(t)$ : Lymphocytes CD4+ sains (cellules cibles).
                \item $I(t)$ : Cellules CD4+ infectées productrices.
                \item $V(t)$ : Virions libres circulant dans le sang.
            \end{itemize}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{table}
                \centering
                \scriptsize
                \caption{Paramètres biomédicaux de référence}
                \begin{tabular}{l c l}
                    \toprule
                    \textbf{Param.} & \textbf{Valeur} & \textbf{Signification} \\
                    \midrule
                    $s$ & $10\text{ j}^{-1}$ & Production CD4+ \\
                    $d$ & $0{,}01\text{ j}^{-1}$ & Mortalité CD4+ ($T_0=1000$) \\
                    $\beta$ & $4{,}6 \times 10^{-5}$ & Infectivité virale \\
                    $\delta$ & $0{,}50\text{ j}^{-1}$ & Mortalité infectées \\
                    $p$ & $1000\text{ j}^{-1}$ & Virions par cellule \\
                    $c$ & $23\text{ j}^{-1}$ & Clairance plasmatique \\
                    \bottomrule
                \end{tabular}
            \end{table}

            \begin{alertebox}[\textbf{Source de Raideur Numérique}]
                \scriptsize
                $c = 23\text{ j}^{-1}$ contre $d = 0{,}01\text{ j}^{-1}$ : ratio d'échelles de temps supérieur à $2000$ !
            \end{alertebox}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Stabilité Globale \& Bifurcation Transcritique}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \begin{succesbox}[\textbf{1. Équilibre Sans Infection (DFE)}]
                \scriptsize
                $$E_0 = (T_0, 0, 0) = (1000, 0, 0)$$
                \textbf{Stabilité Globale (Lyapunov) :}
                Avec $L = I + \frac{\delta}{p}V$, sa dérivée vérifie :
                $$\dot{L} \le \frac{\delta c}{p}(\mathcal{R}_0 - 1)V \le 0 \quad (\text{si } \mathcal{R}_0 \le 1)$$
                $E_0$ est \textbf{globalement asymptotiquement stable}.
            \end{succesbox}
            \vspace{0.08cm}
            \begin{alertebox}[\textbf{2. Équilibre Endémique (EE) : $E^*$}]
                \scriptsize
                Existe si et seulement si $\mathbf{\mathcal{R}_0 > 1}$ :
                $$T^* = \frac{T_0}{\mathcal{R}_0} = 250, \quad V^* = 652{,}2\text{ virions/}\mu\text{L}$$
            \end{alertebox}
        \end{column}

\section{Méthodes Numériques}

\begin{frame}{Conditionnement de la Jacobienne \& Équilibrage LAPACK}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \begin{alertebox}[\textbf{Le Piège du Conditionnement Élevé}]
                \small
                À l'équilibre endémique, la disparité d'échelles ($s=10$ vs $p=1000$) dégrade la Jacobienne :
                $$\kappa_1(J) = \|J\|_1 \cdot \|J^{-1}\|_1 \approx \mathbf{1{,}52 \times 10^5}$$

            \end{alertebox}

            \begin{succesbox}[\textbf{La Solution LAPACK : \texttt{DGEEQU}}]
                \small
                Calcul des facteurs d'échelle optimaux $R$ et $C$ :
                $$\widehat{J} = R \cdot J \cdot C \implies \mathbf{\kappa_1(\widehat{J}) \approx 2{,}4 \times 10^2}$$
                \textbf{Gain de 3 ordres de grandeur sur la précision !}
            \end{succesbox}
        \end{column}

\begin{frame}{Résolution Non-Linéaire : Méthode de Newton avec \texttt{DGESV}}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \begin{messagecle}[\textbf{Algorithme de Newton Vectoriel}]
                \scriptsize
                Recherche de l'état d'équilibre $f(y^*) = 0$ :
                $$J(y_k) \cdot \delta_k = -f(y_k), \quad y_{k+1} = y_k + \delta_k$$
                \begin{itemize}
                    \item Résolu par la routine LAPACK \textbf{\texttt{DGESV}} (LU partiel).
                    \item Aucune inversion matricielle explicite (stabilité numérique).
                \end{itemize}
            \end{messagecle}
            \vspace{0.04cm}
            \begin{succesbox}[\textbf{Convergence Quadratique ($10^{-16}$)}]
                \scriptsize
                Convergence ultra-rapide validée en seulement \textbf{5 à 6 itérations} quel que soit le point initial choisi !
            \end{succesbox}
        \end{column}

        \begin{column}{0.48\textwidth}
